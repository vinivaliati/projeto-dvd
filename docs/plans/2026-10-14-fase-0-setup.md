```python
"""Detect which media tools are installed and their versions."""

import re
import subprocess
from functools import cache

# Version command and version regex for each tool, measured in the Docker image.
# dvdauthor and spumux have no version flag: their help text carries the version on
# stderr and they exit with 1 and 255, so exit codes are ignored.
TOOLS: dict[str, tuple[list[str], str]] = {
    "ffmpeg": (["ffmpeg", "-version"], r"ffmpeg version (\S+)"),
    "ffprobe": (["ffprobe", "-version"], r"ffprobe version (\S+)"),
    "dvdauthor": (["dvdauthor", "--help"], r"dvdauthor, version (\d+(?:\.\d+)*)"),
    "spumux": (["spumux", "--help"], r"spumux, version (\d+(?:\.\d+)*)"),
    "xorriso": (["xorriso", "-version"], r"^xorriso (\d+(?:\.\d+)*)"),
    "tsMuxeR": (["tsMuxeR", "--version"], r"tsMuxeR version (\d+(?:\.\d+)*)"),
}

TIMEOUT_SECONDS = 10


def parse_version(tool: str, output: str) -> str | None:
    """Extract the version of `tool` from its output, or None if it is not there."""
    match = re.search(TOOLS[tool][1], output, re.MULTILINE)
    return match.group(1) if match else None


def _run(command: list[str]) -> str | None:
    """Return stdout and stderr together, or None if the tool cannot run."""
    try:
        result = subprocess.run(
            command,
            capture_output=True,
            text=True,
            errors="replace",
            timeout=TIMEOUT_SECONDS,
            check=False,
        )
    except (OSError, subprocess.TimeoutExpired):
        return None
    return result.stdout + result.stderr


@cache
def detect_versions() -> dict[str, str | None]:
    """Return the version of each media tool, or None when it is missing or unrecognized.

    Cached because versions cannot change while the process runs, and the container
    healthcheck calls this repeatedly.
    """
    versions: dict[str, str | None] = {}
    for tool, (command, _) in TOOLS.items():
        output = _run(command)
        versions[tool] = None if output is None else parse_version(tool, output)
    return versions
```

```python
"""Web application: hello page and health check."""

from pathlib import Path

from fastapi import FastAPI, Request
from fastapi.responses import HTMLResponse, JSONResponse
from fastapi.templating import Jinja2Templates

from projeto_dvd.tools import detect_versions

app = FastAPI(title="projeto-dvd")
templates = Jinja2Templates(directory=Path(__file__).parent / "web" / "templates")


@app.get("/", response_class=HTMLResponse)
def index(request: Request) -> HTMLResponse:
    return templates.TemplateResponse(request, "index.html")


@app.get("/health")
def health() -> JSONResponse:
    """Report the media tool versions; 503 if any tool is missing, so the image is broken."""
    tools = detect_versions()
    ok = all(tools.values())
    return JSONResponse(
        {"status": "ok" if ok else "missing", "tools": tools},
        status_code=200 if ok else 503,
    )
```

````markdown
# projeto-dvd

A local web app that turns a personal movie and TV collection into **authored DVD and Blu-ray
discs with navigable menus**, plus **printable covers** for the cases.

The goal is a physical collection meant to last: discs that follow the official DVD-Video and
BDMV specifications closely enough to play on an ordinary player 20 years from now, with menus
simple enough for a child or a grandparent to use with just the remote.

> **Status:** Phase 0 (project setup). See the roadmap in [`CLAUDE.md`](CLAUDE.md).

## How it works

```
scan → probe → match metadata (TMDB) → disc project (DVD-5/9, BD-25/50) → encode → menu → authoring → ISO
```

The app produces an `.iso` file. Burning happens on Windows (ImgBurn or the built-in burner),
because WSL cannot access the optical drive.

## Stack

- Python 3.12, FastAPI, Jinja2 (HTMX and Tailwind arrive with the first real screens)
- Media toolchain: ffmpeg, dvdauthor/spumux, tsMuxeR, xorriso
- Docker and docker compose; GitHub Actions CI
- uv, ruff, mypy (strict), pytest

## Running with Docker

Requirements: Docker (on Windows, Docker Desktop with WSL integration enabled).

```bash
cp .env.example .env   # then edit the three host paths
make dev               # docker compose up --build
```

Open <http://localhost:8000>. <http://localhost:8000/health> lists the media tool versions found
in the container and returns 503 if any of them is missing.

The media folder is mounted **read-only**: the app never modifies the original files.

## Development

Requirements: [uv](https://docs.astral.sh/uv/).

| Command | What it does |
|---|---|
| `make test` | Run the test suite with coverage (minimum 90%) |
| `make lint` | ruff lint, ruff format check, mypy strict |
| `make format` | Format and auto-fix with ruff |
| `make dev` | Build and start the app with docker compose |

## Maintenance

- The media toolchain is pinned (base image digest, Debian package versions, tsMuxeR checksum).
  When Debian ships a security update the build breaks on purpose; update the pins in the
  `Dockerfile` and commit.
- tsMuxeR has low upstream activity. Keep a copy of
  `tsMuxer-2.7.0-linux.zip` (sha256 `ceaaa181ab70e201685b1e45260d337d1cbdb0aba1408d4fbc47e232a7e4c987`)
  outside GitHub.
````
